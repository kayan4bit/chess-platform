# Replace .github/workflows/ci.yml with this file via the GitHub web editor.
# (OAuth pushes from the Devin proxy can't write workflow files; web editor can.)
#
# This switches the existing CI from pnpm to yarn 1.22.22 to match the
# repo's `packageManager` field. Once pasted, CI on this branch will pass.

name: CI

on:
  push:
    branches: [main]
  pull_request:

jobs:
  build-test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: yarn
      - name: Install yarn 1.22.22
        run: npm install -g yarn@1.22.22
      - name: Install dependencies
        run: yarn install --frozen-lockfile --ignore-engines --network-timeout 600000
      - name: Prisma generate
        working-directory: apps/backend
        run: npx prisma generate
      - name: Backend typecheck
        run: yarn workspace @chess/backend typecheck
      - name: Frontend typecheck
        run: yarn workspace @chess/frontend typecheck
      - name: Backend build
        run: yarn workspace @chess/backend build
      - name: Frontend build
        env:
          NEXT_PUBLIC_API_URL: http://localhost:4000
          NEXT_PUBLIC_WS_URL: http://localhost:4000
        run: yarn workspace @chess/frontend build
      - name: Unit tests
        run: yarn workspace @chess/backend test
